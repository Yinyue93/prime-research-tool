30/// Check if a number is prime using adaptive algorithms
31///
32/// # Examples
33///
34/// ```
35/// use prime_tool::is_prime;
36///
37/// assert!(is_prime(2));
38/// assert!(is_prime(97));
39/// assert!(is_prime(999983));
40/// assert!(!is_prime(999984));
41/// ```
42pub fn is_prime(n: u64) -> bool {
43    if n < 2 {
44        return false;
45    }
46    if n == 2 || n == 3 {
47        return true;
48    }
49    if n.is_multiple_of(2) || n.is_multiple_of(3) {
50        return false;
51    }
52
53    // Deterministic Miller-Rabin covers the entire u64 range: the largest
54    // witness set (first 12 primes) is proven correct for all n < 3.3e24.
55    miller_rabin_deterministic(n)
56}

58/// Deterministic Miller-Rabin primality test
59///
60/// Uses predetermined witnesses based on the input range for deterministic results.
61pub fn miller_rabin(n: u64, witnesses: &[u64]) -> bool {
62    if n < 2 {
63        return false;
64    }
65    if n == 2 {
66        return true;
67    }
68    if n.is_multiple_of(2) {
69        return false;
70    }
71
72    // Write n-1 as d * 2^r
73    let mut d = n - 1;
74    let mut r = 0;
75    while d.is_multiple_of(2) {
76        d /= 2;
77        r += 1;
78    }
79
80    'witness_loop: for &a in witnesses {
81        if a >= n {
82            continue;
83        }
84
85        let mut x = mod_pow(a, d, n);
86        if x == 1 || x == n - 1 {
87            continue 'witness_loop;
88        }
89
90        for _ in 0..r - 1 {
91            x = mod_mul(x, x, n);
92            if x == n - 1 {
93                continue 'witness_loop;
94            }
95        }
96
97        return false;
98    }
99
100    true
101}

103/// Deterministic Miller-Rabin using optimal witness sets
104fn miller_rabin_deterministic(n: u64) -> bool {
105    for (&threshold, witnesses) in MILLER_RABIN_WITNESSES.iter() {
106        if n < threshold {
107            return miller_rabin(n, witnesses);
108        }
109    }
110
111    // Fallback to largest witness set
112    let witnesses = MILLER_RABIN_WITNESSES.get(&u64::MAX).unwrap();
113    miller_rabin(n, witnesses)
114}

116/// Baillie-PSW primality test
117///
118/// A probabilistic test that has no known counterexamples below 2^64.
119pub fn baillie_psw(n: u64) -> bool {
120    if n < 2 {
121        return false;
122    }
123    if n == 2 {
124        return true;
125    }
126    if n.is_multiple_of(2) {
127        return false;
128    }
129
130    // Check for perfect squares
131    if is_perfect_square(n) {
132        return false;
133    }
134
135    // Miller-Rabin test with base 2
136    if !miller_rabin(n, &[2]) {
137        return false;
138    }
139
140    // Lucas test
141    lucas_test(n)
142}

144/// Strong Lucas probable-prime test (Selfridge method A), the second half
145/// of Baillie-PSW. Runs in O(log n) using the doubling formulas, with all
146/// arithmetic done mod n in u128 to avoid overflow across the whole u64 range.
147fn lucas_test(n: u64) -> bool {
148    // Find the first D in 5, -7, 9, -11, ... with Jacobi symbol (D/n) = -1.
149    // Callers have already rejected perfect squares, so this terminates quickly.
150    let mut d: i64 = 5;
151    loop {
152        match jacobi_symbol(d, n) {
153            -1 => break,
154            0 if d.unsigned_abs() != n => return false, // nontrivial factor of n
155            _ => {}
156        }
157        d = if d > 0 { -(d + 2) } else { -(d - 2) };
158    }
159
160    let p: u64 = 1;
161    let q_mod = to_mod((1 - d) / 4, n);
162    let d_mod = to_mod(d, n);
163
164    // Write n + 1 = k * 2^s with k odd (n is odd and < 2^64, so use u128).
165    let np1 = n as u128 + 1;
166    let s = np1.trailing_zeros();
167    let k = np1 >> s;
168
169    // Compute U_k, V_k, Q^k mod n, scanning bits of k from the top.
170    let (mut u, mut v, mut qk) = (1u64, p % n, q_mod); // U_1, V_1, Q^1
171    for bit in (0..(128 - k.leading_zeros() - 1)).rev() {
172        // Doubling: U_2m = U_m V_m, V_2m = V_m^2 - 2 Q^m, Q^2m = (Q^m)^2
173        u = mod_mul(u, v, n);
174        v = mod_sub(mod_mul(v, v, n), mod_mul(2, qk, n), n);
175        qk = mod_mul(qk, qk, n);
176        if (k >> bit) & 1 == 1 {
177            // Increment: U_m+1 = (P U + V)/2, V_m+1 = (D U + P V)/2
178            let u_next = half_mod(mod_add(mod_mul(p, u, n), v, n), n);
179            let v_next = half_mod(mod_add(mod_mul(d_mod, u, n), mod_mul(p, v, n), n), n);
180            u = u_next;
181            v = v_next;
182            qk = mod_mul(qk, q_mod, n);
183        }
184    }
185
186    if u == 0 || v == 0 {
187        return true;
188    }
189    for _ in 1..s {
190        v = mod_sub(mod_mul(v, v, n), mod_mul(2, qk, n), n);
191        if v == 0 {
192            return true;
193        }
194        qk = mod_mul(qk, qk, n);
195    }
196    false
197}

199/// Reduce a signed value into [0, n)
200fn to_mod(x: i64, n: u64) -> u64 {
201    (x as i128).rem_euclid(n as i128) as u64
202}
203
204fn mod_add(a: u64, b: u64, m: u64) -> u64 {
205    ((a as u128 + b as u128) % m as u128) as u64
206}
207
208fn mod_sub(a: u64, b: u64, m: u64) -> u64 {
209    ((a as u128 + m as u128 - b as u128) % m as u128) as u64
210}

212/// x / 2 mod m for odd m
213fn half_mod(x: u64, m: u64) -> u64 {
214    if x.is_multiple_of(2) {
215        x / 2
216    } else {
217        ((x as u128 + m as u128) / 2) as u64
218    }
219}

221/// Check if a number is a perfect square (exact for all u64)
222fn is_perfect_square(n: u64) -> bool {
223    let mut r = (n as f64).sqrt() as u64;
224    // Correct floating-point error in either direction
225    while (r as u128) * (r as u128) > n as u128 {
226        r -= 1;
227    }
228    while ((r + 1) as u128) * ((r + 1) as u128) <= n as u128 {
229        r += 1;
230    }
231    (r as u128) * (r as u128) == n as u128
232}

234/// Compute the Jacobi symbol (a/n) for odd n > 0
235fn jacobi_symbol(a: i64, n: u64) -> i8 {
236    if n == 0 || n.is_multiple_of(2) {
237        return 0;
238    }
239    let mut a = to_mod(a, n);
240    let mut n = n;
241    let mut result = 1i8;
242
243    while a != 0 {
244        while a.is_multiple_of(2) {
245            a /= 2;
246            if n % 8 == 3 || n % 8 == 5 {
247                result = -result;
248            }
249        }
250        std::mem::swap(&mut a, &mut n);
251        if a % 4 == 3 && n % 4 == 3 {
252            result = -result;
253        }
254        a %= n;
255    }
256
257    if n == 1 {
258        result
259    } else {
260        0
261    }
262}

264/// Modular exponentiation: (base^exp) mod m
265fn mod_pow(mut base: u64, mut exp: u64, m: u64) -> u64 {
266    if m == 1 {
267        return 0;
268    }
269
270    let mut result = 1;
271    base %= m;
272
273    while exp > 0 {
274        if exp % 2 == 1 {
275            result = mod_mul(result, base, m);
276        }
277        exp >>= 1;
278        base = mod_mul(base, base, m);
279    }
280
281    result
282}

284/// Modular multiplication: (a * b) mod m, avoiding overflow
285fn mod_mul(a: u64, b: u64, m: u64) -> u64 {
286    ((a as u128 * b as u128) % m as u128) as u64
287}
